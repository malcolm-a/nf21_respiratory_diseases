% listings_setup.tex
% Centralized configuration for code listings (portable: no shell-escape).

% Colors
% (Keep existing neutrals for background/frame/line numbers)
\definecolor{lstbg}{RGB}{248,248,248}
\definecolor{lstframe}{RGB}{220,220,220}
\definecolor{lstnum}{RGB}{120,120,120}

% Palette (from your snippet)
\definecolor{darkred}{rgb}{0.6,0.0,0.0}
\definecolor{darkgreen}{rgb}{0,0.50,0}
\definecolor{lightblue}{rgb}{0.0,0.42,0.91}
\definecolor{orange}{rgb}{0.99,0.48,0.13}
\definecolor{grass}{rgb}{0.18,0.80,0.18}
\definecolor{pink}{rgb}{0.97,0.15,0.45}

% Backwards-compat color aliases (so the rest of the config stays stable)
\colorlet{lstkw}{lightblue}
\colorlet{lstcom}{darkgreen}
\colorlet{lststr}{darkred}

% Base style
\lstdefinestyle{base}{
    basicstyle=\ttfamily\small,
    breaklines=true,
    frame=single,
    rulecolor=\color{lstframe},
    backgroundcolor=\color{lstbg},
    showstringspaces=false,
    numbers=left,
    numberstyle=\scriptsize\color{lstnum},
    tabsize=4,
    columns=fullflexible,
    % Minimal Unicode support for common symbols/accents seen in outputs
    literate={²}{{$^2$}}1 {×}{{$\times$}}1
             {é}{{\'e}}1 {è}{{\`e}}1 {ê}{{\^e}}1 {ë}{{\"e}}1
             {à}{{\`a}}1 {â}{{\^a}}1 {ä}{{\"a}}1
             {î}{{\^\i}}1 {ï}{{\"\i}}1
             {ô}{{\^o}}1 {ö}{{\"o}}1
             {ù}{{\`u}}1 {û}{{\^u}}1 {ü}{{\"u}}1
             {ç}{{\c{c}}}1
             {É}{{\'E}}1 {È}{{\`E}}1 {Ê}{{\^E}}1 {Ë}{{\"E}}1
             {À}{{\`A}}1 {Â}{{\^A}}1 {Ä}{{\"A}}1
             {Î}{{\^I}}1 {Ï}{{\"I}}1
             {Ô}{{\^O}}1 {Ö}{{\"O}}1
             {Ù}{{\`U}}1 {Û}{{\^U}}1 {Ü}{{\"U}}1
             {Ç}{{\c{C}}}1
}

% Extended Python keywords (from your snippet)
\lstdefinelanguage{PythonPlus}[]{Python}{
    morekeywords=[1]{as,assert,nonlocal,with,yield,self,True,False,None},
    morekeywords=[2]{__init__,__add__,__mul__,__div__,__sub__,__call__,__getitem__,__setitem__,__eq__,__ne__,__nonzero__,__rmul__,__radd__,__repr__,__str__,__get__,__truediv__,__pow__,__name__,__future__,__all__},
    morekeywords=[3]{object,type,isinstance,copy,deepcopy,zip,enumerate,reversed,list,set,len,dict,tuple,range,xrange,append,execfile,real,imag,reduce,str,repr},
    morekeywords=[4]{Exception,NameError,IndexError,SyntaxError,TypeError,ValueError,OverflowError,ZeroDivisionError},
    morekeywords=[5]{ode,fsolve,sqrt,exp,sin,cos,arctan,arctan2,arccos,pi,array,norm,solve,dot,arange,isscalar,max,sum,flatten,shape,reshape,find,any,all,abs,plot,linspace,legend,quad,polyval,polyfit,hstack,concatenate,vstack,column_stack,empty,zeros,ones,rand,vander,grid,pcolor,eig,eigs,eigvals,svd,qr,tan,det,logspace,roll,min,mean,cumsum,cumprod,diff,vectorize,lstsq,cla,eye,xlabel,ylabel,squeeze}
}

% Language-specific style
\lstdefinestyle{pythonstyle}{
    style=base,
    language=PythonPlus,
    commentstyle=\color{darkgreen},
    keywordstyle=\color{lightblue}\bfseries,
    keywordstyle=[2]\color{lightblue}\bfseries,
    keywordstyle=[3]\color{grass},
    keywordstyle=[4]\color{darkred},
    keywordstyle=[5]\color{orange},
    stringstyle=\color{darkred}
}

% Plain text style
\lstdefinestyle{textstyle}{
    style=base,
    basicstyle=\ttfamily\small
}

\lstset{style=base}

% Captions / list name
\renewcommand{\lstlistingname}{Code}
\renewcommand{\lstlistlistingname}{Liste des codes sources}
